  [*] --> EnrichmentPending
  EnrichmentPending --> Querying : callWeather()
  Querying --> EnrichmentSuccessful : receive(data)/attachWeather()
  Querying --> Retry : timeout() || error()
  Retry --> Querying : backoffRetry()
  Retry --> EnrichmentFailed : retryExhausted()
}
